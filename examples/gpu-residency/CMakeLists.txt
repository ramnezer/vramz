cmake_minimum_required(VERSION 3.25)
project(VRAMZPublicConsumer LANGUAGES CXX)
find_package(VRAMZ 1.0 CONFIG REQUIRED)
add_executable(vramz-public-demo main.cpp workload.cpp)
target_link_libraries(vramz-public-demo PRIVATE VRAMZ::cuda Threads::Threads)
target_compile_features(vramz-public-demo PRIVATE cxx_std_20)
target_compile_options(vramz-public-demo PRIVATE -Wall -Wextra -Wpedantic -Werror -fstack-protector-strong)
target_link_options(vramz-public-demo PRIVATE -Wl,-z,relro,-z,now,--no-undefined)
set_target_properties(vramz-public-demo PROPERTIES SKIP_BUILD_RPATH TRUE BUILD_RPATH "" INSTALL_RPATH "")
